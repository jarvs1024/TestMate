# 知识库 MCP（`/mcp/kb`）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增独立 MCP 域 `kb`（`/mcp/kb`），提供 7 个读工具与 5 个写工具，让 Agent 通过 MCP 检索知识库、上传/替换文档，并把权限、内置卡片与前端令牌 UI 一起接通。

**Architecture:** 复用现有 `extensions` 域的 `build_mcp` / `ToolSpec` / 配额 / 审计基础设施，新增 `app/mcp/kb/{server,tools,adapter}.py`。写操作统一调用从 `app/api/kb.py` 抽出的 `upload_prepared_documents` / `replace_document`，保证网页端、文本内联、上传票直传三条入口行为一致。二进制文件走「数据集级可复用上传票 + multipart 直传」，不经过 base64。

**Tech Stack:** FastAPI · SQLAlchemy async · MySQL · Redis · FastMCP (Streamable HTTP) · Vue 3 + Element Plus + Vitest

**Spec:** `docs/specs/2026-11-06-kb-mcp-design.md`

## Global Constraints

- 工具总数固定为 **12 个 = 7 读 + 5 写**；读工具 `scope="kb:read"`、`read_only=True`；写工具 `scope="kb:write"`、`read_only=False`。
- 上传 / 替换 / 重解析工具返回的 `ok=true`、`state=submitted` **只表示已提交并成功发起解析**，绝不表示解析成功；每篇文档带提交时的 `run`，解析结果只能通过 `kb_document_status` 查询。
- 上传票 `KB_UPLOAD_TICKET_TYPE = "kb-upload"`、`KB_UPLOAD_TICKET_TTL = 1800`；绑定用户 + 数据集，30 分钟内可复用；放请求头 `X-KB-Upload-Ticket`，不放进 URL。
- 票据端点必须**先验票、再读 body**（`await request.form(...)` 之前完成所有票据校验）；票的有效期按请求开始时刻计算。
- 不新增数据库迁移；`McpApiToken.scopes` / `McpOAuthGrant.scopes` 是 JSON 列，直接写入 `kb:read` / `kb:write`。
- 不新增 nginx 配置、前端路由、页面入口；知识库通过扩展库内置卡片接入。
- 不暴露删除类工具、`source_url` 拉取、base64 二进制入参。
- REST 行为除「MCP 令牌不能再当登录令牌」这一条安全收紧外保持不变。
- 所有业务错误只透出代码里写死的固定文案；502 / RAGFlow 内部错误统一转 `kb_unavailable`，不得泄露 base URL、API key 或异常原文。
- 执行阶段**不自动 git commit**；每个任务以测试通过作为检查点，提交仅在用户明确要求时进行。

```python
    """按部署地址现拼 MCP 配置 —— base 是平台从外面访问的根地址."""
```

```python
# ===== 知识库上传票据 =====

#: 上传票是**数据集级、可复用**的: 批量上传 / 慢速内网不能因为一张短票
#: 变成「前 3 个成功、后 2 个 401」。30 分钟覆盖一批 5×20MB 的常规耗时。
KB_UPLOAD_TICKET_TYPE = "kb-upload"
KB_UPLOAD_TICKET_TTL = 1800


def create_kb_upload_ticket(
    *,
    user_id: int,
    token_version: int,
    dataset_id: str,
    ttl_seconds: int = KB_UPLOAD_TICKET_TTL,
) -> tuple[str, int]:
    """签发绑定用户 + 数据集的短时上传票，返回 (ticket, expires_in_seconds)。"""
    expire = datetime.utcnow() + timedelta(seconds=ttl_seconds)
    payload: dict[str, Any] = {
        "typ": KB_UPLOAD_TICKET_TYPE,
        "uid": int(user_id),
        "tv": int(token_version),
        "ds": dataset_id,
        "exp": expire,
    }
    ticket = jwt.encode(payload, settings.JWT_SECRET, algorithm=settings.JWT_ALGORITHM)
    return ticket, ttl_seconds


def decode_kb_upload_ticket(ticket: str) -> dict[str, Any] | None:
    """只接受签名有效、未过期且类型为 kb-upload 的票据。"""
    if not ticket:
        return None
    try:
        payload = jwt.decode(
            ticket,
            settings.JWT_SECRET,
            algorithms=[settings.JWT_ALGORITHM],
            options={"require_exp": True},
        )
    except (JWTError, TypeError, ValueError):
        return None
    if payload.get("typ") != KB_UPLOAD_TICKET_TYPE:
        return None
    return payload
```

```python
@router.post("/datasets/{dataset_id}/documents/ticket-upload")
async def ticket_upload(dataset_id: str, request: Request) -> dict:
    """上传票直传 —— 先验票，再读 multipart；一张票可复用于整批。"""
    claims = decode_kb_upload_ticket(request.headers.get("X-KB-Upload-Ticket", ""))
    if not claims:
        raise HTTPException(status_code=401, detail="上传票无效或已过期")
    if str(claims.get("ds") or "") != dataset_id:
        raise HTTPException(status_code=403, detail="上传票与数据集不匹配")
    try:
        user_id = int(claims.get("uid") or 0)
        token_version = int(claims.get("tv") or 0)
    except (TypeError, ValueError):
        raise HTTPException(status_code=401, detail="上传票无效")

    async with AsyncSessionLocal() as session:
        user = await session.get(User, user_id)
    if not user or not user.is_active:
        raise HTTPException(status_code=403, detail="上传票对应账号不可用")
    if token_version != (user.token_version or 0):
        raise HTTPException(status_code=403, detail="上传票已失效, 请重新签票")
    if not can_write_kb(user):
        raise HTTPException(status_code=403, detail="只读账号不能上传文档")

    try:
        form = await request.form(
            max_files=MAX_UPLOAD_FILES, max_fields=MAX_UPLOAD_FILES
        )
    except Exception:
        raise HTTPException(status_code=422, detail="上传表单格式错误")
    try:
        uploads = form.getlist("file")
        replace_id = str(form.get("document_id") or "").strip()
        if not uploads:
            raise HTTPException(status_code=422, detail="没有选择文件")
        if replace_id:
            if len(uploads) != 1:
                raise HTTPException(status_code=422, detail="替换模式一次只能传 1 个文件")
            upload = uploads[0]
            raw_name = (getattr(upload, "filename", "") or "").strip()
            payload = await _read_upload_limited(
                upload, MAX_UPLOAD_BYTES, raw_name or "document"
            )
            return await replace_document(
                dataset_id, replace_id, raw_name, payload,
                getattr(upload, "content_type", "") or "application/octet-stream",
                user,
            )

        prepared: list[tuple[str, bytes, str]] = []
        for f in uploads:
            raw_name = (getattr(f, "filename", "") or "").strip()
            if not raw_name:
                continue
            name = _sanitize_kb_filename(raw_name, f"doc-{len(prepared) + 1}")
            data = await _read_upload_limited(f, MAX_UPLOAD_BYTES, name)
            if not data:
                continue
            prepared.append(
                (name, data, getattr(f, "content_type", "") or "application/octet-stream")
            )
        if not prepared:
            raise HTTPException(status_code=422, detail="没有可上传的文件 (空文件)")
        return await upload_prepared_documents(dataset_id, prepared, user)
    finally:
        await form.close()
```

```python
"""Adapters from authenticated MCP calls to knowledge-base core functions."""
from __future__ import annotations

import asyncio
import logging
import os
import time
from typing import Any

from fastapi import HTTPException

from app.api import kb
from app.mcp.common.context import McpContext
from app.mcp.common.errors import McpRuntimeError

logger = logging.getLogger(__name__)

TEXT_DOC_EXTENSIONS = {
    ".txt", ".md", ".markdown", ".csv", ".json", ".log", ".html", ".htm",
}
MAX_MCP_INLINE_BYTES = 2 * 1024 * 1024
MAX_MCP_BATCH_DOCS = 5
MAX_MCP_BATCH_BYTES = 8 * 1024 * 1024

#: 测试可替换的时钟/睡眠，避免真的等 2 秒。
_monotonic = time.monotonic
_sleep = asyncio.sleep


def _business_error(exc: HTTPException) -> McpRuntimeError:
    code = {
        403: "permission_denied",
        404: "not_found",
        409: "conflict",
        413: "file_too_large",
        422: "invalid_arguments",
    }.get(exc.status_code)
    if code is None:
        return McpRuntimeError("kb_unavailable", "知识库服务暂时不可用，请稍后重试。")
    detail = exc.detail if isinstance(exc.detail, str) else "请求失败"
    return McpRuntimeError(code, detail)


async def _call(operation):
    try:
        return await operation
    except McpRuntimeError:
        raise
    except HTTPException as exc:
        raise _business_error(exc) from None
    except Exception:
        logger.exception("kb mcp adapter failure")
        raise McpRuntimeError("kb_unavailable", "知识库服务暂时不可用，请稍后重试。") from None


async def _require_dataset(dataset_id: str) -> dict[str, Any]:
    page = 1
    while page <= 5:
        items = await _call(kb._dataset_items(page=page, page_size=100))
        for item in items:
            if str(item.get("id") or "") == dataset_id:
                return item
        if len(items) < 100:
            break
        page += 1
    raise McpRuntimeError("not_found", "数据集不存在")


async def dataset_list(ctx: McpContext, *, page=1, page_size=50, keyword=""):
    if keyword:
        all_items = await _call(
            kb._dataset_items(page=1, page_size=500, keyword=keyword)
        )
        start = max(page - 1, 0) * page_size
        items = all_items[start:start + page_size]
        has_more = start + page_size < len(all_items)
    else:
        items = await _call(kb._dataset_items(page=page, page_size=page_size))
        has_more = len(items) == page_size
    return {"items": items, "page": page, "page_size": page_size, "has_more": has_more}


async def document_list(
    ctx: McpContext,
    *,
    dataset_id,
    page=1,
    page_size=30,
    orderby="create_time",
    desc=True,
    keywords="",
    run="",
):
    return await _call(
        kb.get_documents(
            dataset_id=dataset_id, page=page, page_size=page_size, orderby=orderby,
            desc=desc, keywords=keywords, run=run, _user=ctx.principal.user,
        )
    )


async def document_search(
    ctx: McpContext, *, keywords, dataset_id="", limit=200
):
    return await _call(

        )
    )

async def _status_document(dataset_id: str, document_id: str, name: str) -> dict:
    if name:
        data = await _call(
            kb.list_documents(dataset_id=dataset_id, page=1, page_size=100, keywords=name)
        )
        for doc in data.get("docs") or []:
            if doc.get("name") == name or str(doc.get("id") or "") == document_id:
                return doc
    if not document_id:
        raise McpRuntimeError("invalid_arguments", "document_id 或 name 至少提供一个")
    doc = await _call(kb._find_document(dataset_id, document_id))
    if not doc:
        raise McpRuntimeError("not_found", "文档不存在")
    return doc


async def document_status(
    ctx: McpContext,
    *,
    dataset_id,
    document_id="",
    name="",
    wait_seconds=0,
):
    deadline = _monotonic() + max(int(wait_seconds), 0)
    while True:
        doc = await _status_document(dataset_id, document_id, name)
        run = str(doc.get("run") or "UNSTART")
        finished = run in {"DONE", "FAIL", "CANCEL"}
        timed_out = wait_seconds > 0 and not finished and _monotonic() >= deadline
        if finished or timed_out or wait_seconds <= 0:
            return {
                "document_id": str(doc.get("id") or document_id),
                "name": str(doc.get("name") or name),
                "run": run,
                "progress": doc.get("progress", 0.0),
                "progress_msg": doc.get("progress_msg", ""),
                "chunk_count": doc.get("chunk_count", 0),
                "token_count": doc.get("token_count", 0),
                "timed_out": timed_out,
            }
        await _sleep(2)


async def search(
    ctx: McpContext,
    *,
    question,
    dataset_ids,
    document_ids=None,
    top_k=5,
    similarity_threshold=0.2,
    vector_similarity_weight=0.3,
    keyword=False,
    highlight=True,
    page_size=10,
):
    payload = kb.SearchIn(
        question=question,
        dataset_ids=dataset_ids,
        document_ids=document_ids or [],
        top_k=top_k,
        similarity_threshold=similarity_threshold,
        vector_similarity_weight=vector_similarity_weight,
        keyword=keyword,
        highlight=highlight,
        page_size=page_size,
    )
    return await _call(kb.search(payload=payload, _user=ctx.principal.user))


async def health(ctx: McpContext):
    status, message = await _call(kb.probe())
    return {"status": status, "message": message}
```

```python
def _upload_items(name, content, content_type, documents):
    if documents is not None:
        if not 1 <= len(documents) <= MAX_MCP_BATCH_DOCS:
            raise McpRuntimeError("invalid_arguments", "documents 需为 1..5 篇")
        return [dict(item) for item in documents]
    if name is None or content is None:
        raise McpRuntimeError("invalid_arguments", "请提供 name+content 或 documents 数组")
    item = {"name": name, "content": content}
    if content_type is not None:
        item["content_type"] = content_type
    return [item]


def _prepare_text(name: str, content: str, content_type: str | None):
    clean = kb._sanitize_kb_filename(name, "document")
    ext = os.path.splitext(clean.lower())[1]
    if ext not in TEXT_DOC_EXTENSIONS:
        raise McpRuntimeError(
            "invalid_document_type",
            "文本工具只支持 .txt/.md/.markdown/.csv/.json/.log/.html/.htm；"
            "二进制文件请用 kb_document_write_ticket 签票后直传。",
        )
    payload = (content or "").encode("utf-8")
    if not payload:
        raise McpRuntimeError("invalid_arguments", "文件内容是空的")
    if len(payload) > MAX_MCP_INLINE_BYTES:
        raise McpRuntimeError(
            "file_too_large", "单篇文本超过 2 MB，请改用 kb_document_write_ticket 签票后直传。"
        )
    return clean, payload, content_type or "text/plain; charset=utf-8"


def _write_result(
    docs, errors, ingest_started, warnings, *, count_key="uploaded", replaced=None
):
    state = "submitted" if not errors else ("failed" if not docs else "partial")
    result = {
        "ok": not errors,
        "state": state,
        count_key: len(docs),
        "failed": len(errors),
        "ingest_started": ingest_started,
        "warning": " ".join(warnings),
        "docs": docs,
        "errors": errors,
    }
    if replaced is not None:
        result["replaced"] = replaced
    return result

async def document_upload(
    ctx: McpContext,
    *,
    dataset_id,
    name=None,
    content=None,
    content_type=None,
    documents=None,
):
    items = _upload_items(name, content, content_type, documents)
    await _require_dataset(dataset_id)

    prepared = []
    errors = []
    for index, item in enumerate(items):
        label = str(item.get("name") or f"doc-{index + 1}")
        try:
            prepared.append(
                (index, _prepare_text(label, str(item.get("content") or ""), item.get("content_type")))
            )
        except McpRuntimeError as exc:
            errors.append({"name": label, "error": exc.code, "message": exc.message})
    if sum(len(payload) for _, (_, payload, _) in prepared) > MAX_MCP_BATCH_BYTES:
        raise McpRuntimeError(
            "file_too_large", "本次文本总量超过 8 MB，请分批上传或改用上传票。"
        )

    docs, warnings, ingest_started = [], [], False
    for _, file_tuple in prepared:
        try:
            result = await _call(
                kb.upload_prepared_documents(dataset_id, [file_tuple], ctx.principal.user)
            )
        except McpRuntimeError as exc:
            errors.append(
                {"name": file_tuple[0], "error": exc.code, "message": exc.message}
            )
            continue
        ingest_started = ingest_started or bool(result.get("ingest_started"))
        if result.get("warning"):
            warnings.append(str(result["warning"]))
        for doc in result.get("docs") or []:
            docs.append(
                {
                    "document_id": doc.get("id"),
                    "name": doc.get("name"),
                    "state": "submitted",
                    "run": doc.get("run", "UNSTART"),
                    "size": doc.get("size", 0),
                }
            )
    return _write_result(docs, errors, ingest_started, warnings)


async def document_update(
    ctx: McpContext,
    *,
    dataset_id,
    document_id=None,
    content=None,
    documents=None,
):
    if documents is not None:
        if not 1 <= len(documents) <= MAX_MCP_BATCH_DOCS:
            raise McpRuntimeError("invalid_arguments", "documents 需为 1..5 篇")
        items = [dict(item) for item in documents]
    else:
        if document_id is None or content is None:
            raise McpRuntimeError(
                "invalid_arguments", "请提供 document_id+content 或 documents 数组"
            )
        items = [{"document_id": document_id, "content": content}]
    await _require_dataset(dataset_id)

    docs, errors, warnings = [], [], []
    replaced_ids: list[str] = []
    ingest_started = False
    for item in items:
        old_id = str(item.get("document_id") or "")
        try:
            old = await _call(kb._find_document(dataset_id, old_id))
            if not old:
                raise McpRuntimeError("not_found", "文档不存在")
            name, payload, ctype = _prepare_text(
                str(old.get("name") or ""), str(item.get("content") or ""), None
            )
            result = await _call(
                kb.replace_document(
                    dataset_id, old_id, name, payload, ctype, ctx.principal.user
                )
            )
        except McpRuntimeError as exc:
            errors.append(
                {"document_id": old_id, "name": "", "error": exc.code, "message": exc.message}
            )
            continue
        ingest_started = ingest_started or bool(result.get("ingest_started"))
        if result.get("warning"):
            warnings.append(str(result["warning"]))
        updated = result.get("updated") or {}
        docs.append(
            {
                "document_id": updated.get("id"),
                "name": updated.get("name"),
                "state": "submitted",
                "run": updated.get("run", "UNSTART"),
                "size": updated.get("size", 0),
            }
        )
        replaced_ids.append(old_id)

    replaced = replaced_ids[0] if len(items) == 1 and replaced_ids else replaced_ids
    return _write_result(
        docs, errors, ingest_started, warnings, count_key="updated", replaced=replaced
    )
```
